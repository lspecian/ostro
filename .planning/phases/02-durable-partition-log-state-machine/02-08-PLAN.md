```rust
// replicated.rs
#[derive(Copy, Clone, Debug, PartialEq, Eq)] pub struct AppendReceipt { pub base: Offset, pub last: Offset, pub epoch: Epoch }
#[derive(Clone, Debug, PartialEq, Eq)] pub struct Isr(pub Vec<u64> /* node ids */);
#[derive(Clone, Debug, PartialEq, Eq)] pub enum Role { Leader { epoch: Epoch }, Follower { epoch: Epoch }, Offline }
#[derive(Clone, Debug, PartialEq, Eq)] pub enum LogHealth { Healthy, Failed { reason: String } }
#[derive(Clone, Debug, Default, PartialEq, Eq)] pub struct EpochHistory(pub Vec<(Epoch, Offset)>); // impl latest() -> Option<Epoch>, end_offset_for_epoch(e) -> Option<Offset>
pub trait ReplicatedLog: Send + Sync + 'static {
  fn partition(&self) -> PartitionId;
  fn append(&self, epoch: Epoch, cmds: Vec<Bytes>) -> impl Future<Output = Result<AppendReceipt, LogError>> + Send; // returns once position is fixed (enqueued), NOT durable
  fn wait_committed(&self, last: Offset) -> impl Future<Output = Result<(), LogError>> + Send; // Ok only when HW > last
  fn high_watermark(&self) -> watch::Receiver<Offset>; // exclusive
  fn log_end(&self) -> Offset;   // exclusive
  fn log_start(&self) -> Offset;
  fn read(&self, from: Offset, upto: Offset, max_bytes: usize) -> impl Future<Output = Result<Vec<VerifiedBatch>, LogError>> + Send; // batches fully inside [from, upto); from must be a batch boundary
  fn epochs(&self) -> EpochHistory;
  fn isr(&self) -> Isr;
  fn role(&self) -> watch::Receiver<Role>;
  fn health(&self) -> watch::Receiver<LogHealth>;
  fn truncate_to(&self, end: Offset) -> impl Future<Output = Result<Offset, LogError>> + Send;
  fn trim_prefix(&self, permit: TrimPermit) -> impl Future<Output = Result<Offset /* new log_start */, LogError>> + Send;
}
// hw.rs
#[derive(Copy, Clone, Debug)] pub struct ReplicaProgress { pub node: u64, pub leo: Offset, pub in_isr: bool }
pub fn compute_hw(progress: &[ReplicaProgress]) -> Offset; // min leo over in_isr replicas; panics/debug_asserts if none in ISR
// barrier.rs
#[derive(Clone)] pub struct ReadyBarrier { .. } // watch<bool>
impl ReadyBarrier { pub fn new() -> Self; pub fn open(&self); pub fn is_open(&self) -> bool; pub async fn wait_open(&self); pub fn mint(&self, floor: Offset) -> Result<TrimPermit, LogError /* NotReady */>; }
pub struct TrimPermit { floor: Offset } // no public constructor; impl floor(&self) -> Offset
// node.rs
#[derive(Clone, Debug)] pub struct LogNodeConfig { pub wal_dir: PathBuf, pub partitions: PartitionCount, pub node_id: u64, pub wal: GroupCommitConfig, pub max_envelope_bytes: usize, pub max_inflight_appends: usize /* default 1024 */ }
pub struct LogNode { .. } // owns Arc<GroupCommitWal>; impl shutdown()
impl LogNode { pub async fn open(cfg: LogNodeConfig, floors: &[Offset]) -> Result<(LogNode, Vec<Result<Arc<Rf1Log>, LogError>>), LogError>; } // floors.len() == P; per-partition recovery errors are returned per slot, node-level errors as Err
// rf1.rs
pub struct Rf1Log { .. } // impl ReplicatedLog; extra: pub fn recovery(&self) -> &RecoveryReport; pub fn set_role(&self, role: Role)
```
`Rf1Log::append` follows the 02-RESEARCH Pattern 5 sketch: bounded by a `Semaphore(max_inflight_appends)`, per-partition `tokio::sync::Mutex<SeqState{next, current_epoch}>`, `health` checked first, `epoch < current_epoch` → `StaleEpoch`, envelope built with `ostro_types::envelope::encode`, an envelope larger than `max_envelope_bytes` returns `LogError::Wal("envelope too large")` without touching sequencer state or health (callers split batches), `append_enqueued(.., acks = 1)`, then (on Ok only) update epoch history + `next`. After releasing the lock, spawn a task that awaits the CommitWait and on Err calls `self.fail(reason)` (health → Failed, role → Offline). Initial `next = recovery.leo`, initial HW = recovery.leo (recovery fsynced every retained segment), initial epochs from `recovery.epochs`.
