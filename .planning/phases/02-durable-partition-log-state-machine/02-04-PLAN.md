```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)] pub struct BlobRef { pub sha256: [u8; 32], pub len: u64 }
#[derive(Debug, thiserror::Error)] pub enum BlobError { #[error("io: {0}")] Io(#[from] std::io::Error), #[error("blob {0} not found")] NotFound(String), #[error("blob {hex} corrupt: expected sha256 {hex}, got {actual}")] ShaMismatch { hex: String, actual: String }, #[error("blob {hex} length {actual} != {expected}")] LenMismatch { hex: String, expected: u64, actual: u64 } }
pub struct BlobStore { /* root: PathBuf (the data dir) */ }
impl BlobStore {
  pub fn open(data_dir: &Path) -> Result<Self, BlobError>;      // creates blobs/ and tmp/, deletes stale tmp/* files, fsyncs created dirs
  pub fn put(&self, bytes: &[u8]) -> Result<BlobRef, BlobError>; // sync API; callers in async code use spawn_blocking
  pub fn get(&self, r: &BlobRef) -> Result<Vec<u8>, BlobError>;  // verifies len + sha256
  pub fn contains(&self, r: &BlobRef) -> bool;
  pub fn path_of(&self, r: &BlobRef) -> PathBuf;
}
pub fn hex(sha: &[u8; 32]) -> String;
```
