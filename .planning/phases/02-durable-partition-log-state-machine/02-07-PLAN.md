<fjall_adaptation>
Rule condition 2 in the research reads the durable checkpoint from redb. With fjall the floor is the state's own `applied_offset` (read by the caller before log recovery and passed in as `floor: Offset`). This is valid and stronger: the state machine only ever applies commands at offsets below HW, and at RF=1 HW only advances after the WAL fsync, so every offset < applied_offset was fsynced in the WAL. If the state store is missing (rebuild), the floor is `Offset(0)`.
</fjall_adaptation>

// error.rs
#[derive(Debug, thiserror::Error)] pub enum LogError {
  #[error("partition {partition} corrupt in {segment} at byte {byte_pos} (offsets {first}..={last}): {reason}")] Corrupt { partition: u16, segment: PathBuf, byte_pos: u64, first: i64, last: i64, reason: Corruption },
  #[error("stale epoch: current {current:?}")] StaleEpoch { current: Epoch },
  #[error("partition failed: {0}")] Failed(String),
  #[error("io: {0}")] Io(#[from] std::io::Error),
  #[error("wal: {0}")] Wal(String),
  #[error("not ready")] NotReady,
  #[error("offset {0:?} out of range")] OutOfRange(Offset),
  #[error("truncated")] Truncated }
// verify.rs
#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)] pub enum Corruption { FrameCrc{stored: u32, computed: u32}, NotV2, Misplaced{partition: i32, base: i64}, Envelope(EnvelopeError), TrailerMismatch, Gap{expected: u64, found: u64}, EpochRegression{prev: u32, found: u32}, Unparseable, Incomplete }
#[derive(Debug, Clone)] pub struct VerifiedBatch { pub epoch: Epoch, pub base: Offset, pub commands: Vec<Bytes> } // last(), end() (= last+1), commands_with_offsets()
pub fn verify_record(rec: &WalRecord, expect_base: Offset, pid: PartitionId) -> Result<VerifiedBatch, Corruption>;
// recovery.rs
pub struct RecoveryReport { pub segments: Vec<PathBuf>, pub log_start: Offset, pub leo: Offset, pub epochs: Vec<(Epoch, Offset)> /* epoch → first offset */, pub torn: Option<TornCut>, pub records: u64, pub bytes: u64 }
pub struct TornCut { pub segment: PathBuf, pub byte_pos: u64, pub discarded_records: u64, pub reason: Corruption }
pub fn segment_dir(wal_dir: &Path, pid: PartitionId) -> PathBuf; // <wal_dir>/p/<pid>
pub fn list_segments(dir: &Path, pid: PartitionId) -> io::Result<Vec<(u64, PathBuf)>>; // sorted by seg id, includes empty files
pub fn recover_partition(wal_dir: &Path, pid: PartitionId, floor: Offset) -> Result<RecoveryReport, LogError>; // sync fn; callers use spawn_blocking
