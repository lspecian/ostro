Build the SSRF-hardened egress core (FX-08):
1. the `url`-normalised URL gate;
2. the IANA classifier with mapped/NAT64/6to4 unwrap;

pub struct EgressConfig { pub allow_private: Vec<ipnet::IpNet>, pub connect_timeout: Duration /* 5s */,
    pub default_timeout: Duration /* 30s */, pub max_response_bytes: u64 /* 4 MiB */, pub hard_max_response_bytes: u64 /* 64 MiB */,
    pub max_redirects: u8 /* 5 */, pub rate_max_wait: Duration /* 30s */, pub idempotency_window: Duration /* 24h */,
    pub safety_margin: Duration /* 5m */ }
pub enum Denied { Parse(String), Scheme, Userinfo, Port, Placeholder, NotAllowed, IpLiteral(IpAddr), Blocked(IpAddr), Dns(String) }
impl Denied { pub fn reason(&self) -> &'static str /* snake_case variant: parse|scheme|userinfo|port|placeholder|not_allowed|ip_literal|blocked|dns */;
  pub fn to_failure(&self) -> ostro_types::journal::Failure /* NonRetryable egress_denied; details JSON {"reason": reason(), "ip"?: ".."}; message names the rule, never secrets */ }
pub struct CheckedUrl { pub url: url::Url, pub origin: Origin, pub rule: AllowRule }
#[async_trait] pub trait Lookup: Send + Sync { async fn lookup(&self, host: &str) -> std::io::Result<Vec<IpAddr>>; }
