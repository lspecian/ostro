```rust
pub const RUN_INDEX_KEYSPACE: &str = "x_run_index";
pub const RUN_INDEX_VERSION: u32 = 1;
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum IndexKind { Tag = 0x01, Workflow = 0x02, Status = 0x03, Tenant = 0x04, All = 0x05 }
/// Ordering suffix (W2): `!created_at_ms` (u64 BE, bit-inverted) ‖ `!run_uuid` (bit-inverted). Child run ids are uuid v8 (D-P9-7),
/// so the uuid alone does not order by time; created_at_ms does, the uuid breaks ties.
pub fn index_key(kind: IndexKind, value: &[u8], created_at_ms: u64, run_uuid: &[u8; 16]) -> Vec<u8>; // kind ‖ value ‖ 0x00 ‖ sfx (Status: kind ‖ status u8 ‖ sfx; All: kind ‖ sfx)
pub fn index_prefix(kind: IndexKind, value: &[u8]) -> Vec<u8>;
pub fn run_index_rows(run: &RunId, rec: &RunRecord) -> Vec<Vec<u8>>;      // the single source of truth used by hooks, rebuild and the proptest
pub enum IndexState { Ready, Rebuilding }

// crates/ostro-partition/src/store/batch.rs
impl ApplyBatch<'_> { pub(crate) fn insert_derived(&mut self, key: Vec<u8>); pub(crate) fn remove_derived(&mut self, key: Vec<u8>); }

// crates/ostro-partition/src/query.rs
pub struct IndexPage { pub runs: Vec<(RunId, u64 /* created_at_ms */)>, pub next: Option<Vec<u8>> /* last key: carries created_at_ms + uuid */, pub scanned: usize }
pub fn index_page(r: &StoreReader, prefix: &[u8], after: Option<&[u8]>, limit: usize, budget: usize)
    -> Result<IndexPage, IndexError>;   // IndexError::Rebuilding when the marker is missing/stale
pub fn index_state(r: &StoreReader) -> IndexState;
```
Statuses indexed: Failed, Completed, Cancelled, Terminated, TimedOut, ContinuedAsNew (terminal only). Keys hold no value bytes (empty value), except the marker `0x00 'v'` → `{index_version: u32 BE (= RUN_INDEX_VERSION), built_at_applied_offset: u64 BE}` (W3). Every apply batch that writes derived rows also rewrites the marker with that batch's applied offset; at open, a version mismatch or `built_at_applied_offset != applied_offset` means the index may be stale (e.g. a crash between keyspace writes on a store without cross-keyspace atomicity, or a downgrade that applied commands without the hooks) and triggers a rebuild. Values never contain user payloads. Tag/workflow/tenant strings never contain 0x00: the API and apply already bound them; if one does, the hook skips the row and the proptest documents it.
