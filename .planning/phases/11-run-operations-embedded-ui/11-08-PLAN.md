```rust
#[derive(Clone, Debug, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
pub struct DlqRow {
    pub hook_run: Option<[u8; 18]>,
    pub dismissed_at_ms: Option<u64>,
    pub dismiss_reason: Option<String>,
    pub requeued_to: Option<[u8; 18]>,   // restart / rerun target

    pub requeue_mode: Option<u8>,        // 0 resume, 1 restart, 2 rerun
    pub episode_completed_at_ms: u64,    // W5: the failure episode (RunRecord.completed_at_ms) these marks belong to
}
// Field order is the postcard format: never reorder; future fields need a row migrator (D18).

// crates/ostro-partition/src/apply/dlq.rs
pub(crate) fn apply_dlq_update(b: &mut ApplyBatch<'_>, cmd: &DlqUpdate, ..) -> ApplyOutcome;
// Rules:
// - run must exist (RunNotFound otherwise) and be on this partition (WrongPartition, existing check).
// - HookStarted / Dismissed: status must be Failed or TimedOut, else Invalid(BadDlqUpdate).
// - Requeued{mode=0 resume}: accepted for any status (the resume already moved the run out of Failed); records the mark only.
// - Requeued{mode=1|2}: status must be Failed or TimedOut; new_run_id must be 18 bytes.
// - Episode (W5): cmd.episode_completed_at_ms must equal the run's current completed_at_ms for HookStarted/Dismissed/Requeued{1,2}
//   (else Invalid(BadDlqUpdate)); if the stored row's episode differs (an older failure), the row is reset to a fresh DlqRow for the
//   new episode before merging, so marks of an earlier failure never hide a later one.
// - Merge: HookStarted sets hook_run if None (a second HookStarted with a different run → Invalid(BadDlqUpdate));
//   Dismissed sets dismissed_*; Requeued sets requeued_*. Identical re-application → Applied with no change.
// - Never reads payloads or blobs; never touches the runs row.
```
