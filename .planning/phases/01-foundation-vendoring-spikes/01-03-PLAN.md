Verified-compiling host-side pattern for the resource step under wasmtime 50.0.2 bindgen (from the research prototype):
```rust
use wasmtime::component::{bindgen, Component, Linker, Resource, ResourceTable};
bindgen!({ path: "../wit", world: "spike" });
use ostro::spike::journal::{self, StepStart};
pub struct StepRep { seq: u64 }
struct Host { table: ResourceTable, limits: wasmtime::StoreLimits, next: u64, rng: u64, step_drops: u64 }
impl journal::HostStep for Host {
    fn seq(&mut self, s: Resource<journal::Step>) -> u64 { let r: Resource<StepRep> = Resource::new_borrow(s.rep()); self.table.get(&r).unwrap().seq }
    fn complete(&mut self, _s: Resource<journal::Step>, _o: String) -> bool { true }
    fn drop(&mut self, s: Resource<journal::Step>) -> wasmtime::Result<()> { let r: Resource<StepRep> = Resource::new_own(s.rep()); self.table.delete(r)?; self.step_drops += 1; Ok(()) }
}
impl journal::Host for Host {
    fn begin_step(&mut self, _n: String) -> StepStart { self.next += 1; let r = self.table.push(StepRep { seq: self.next }).unwrap(); StepStart::Execute(Resource::new_own(r.rep())) }
    fn begin_step_u64(&mut self, _n: String) -> u64 { self.next += 1; self.next }
    fn complete_u64(&mut self, _h: u64, _o: String) -> bool { true }
}
impl ostro::spike::context::Host for Host { fn now_ms(&mut self) -> u64 { 1_700_000_000_000 } fn random_u64(&mut self) -> u64 { /* seeded LCG or splitmix from --seed */ } }
struct D; impl wasmtime::component::HasData for D { type Data<'a> = &'a mut Host; }
// Spike::add_to_linker::<Host, D>(&mut linker, |h| h)?;
```
Python bindings (componentize-py 0.25.1): `import wit_world; from wit_world.imports import journal, context; class WitWorld(wit_world.WitWorld): ...`; resources are classes usable as context managers (`with start.value as step:`); variants are `journal.StepStart_Replayed` / `journal.StepStart_Execute`.
TS bindings (jco 1.42): `import { beginStep } from 'ostro:spike/journal@0.1.0'`; variants are `{tag: 'execute', val: step}`; exports are camelCase functions (`export function run(input) {...}`, `export function probeRandom() {...}`).
