```rust
pub trait StateMachine: Send + 'static {
    fn apply(&mut self, index: u64, term: u64, command: &[u8]) -> Vec<u8>;
    fn snapshot(&self) -> Vec<u8>;
    fn restore(&mut self, index: u64, data: &[u8]) -> Result<(), String>;
}
pub trait Outbox: Send + 'static { fn send(&mut self, messages: Vec<Message>); }
impl DriverConfig { pub fn new(id: u64) -> Self }
impl RaftDriver {
    pub fn start<S: StateMachine, O: Outbox>(dir: impl Into<PathBuf>, config: DriverConfig, initial_voters: &[u64], state_machine: S, outbox: O) -> io::Result<Self>;
    pub fn handle(&self) -> RaftHandle;
    pub fn stop(self);
}
impl RaftHandle {
    pub async fn wait_for_leader(&self) -> Result<u64, ProposeError>;
    pub async fn propose(&self, data: Vec<u8>) -> Result<Applied, ProposeError>; // Applied { index, term, result: Vec<u8> }
    pub fn status(&self) -> RaftStatus;
}
```
