DevClock contract (write exactly):
```rust
pub struct DevClock { base: Arc<dyn ostro_partition::clock::Clock>, offset_ms: AtomicU64, mode: DevTimeMode, tx: watch::Sender<u64> }
impl DevClock {
    pub fn new(base: Arc<dyn ostro_partition::clock::Clock>, mode: DevTimeMode) -> Arc<Self>;
    pub fn offset_ms(&self) -> u64; pub fn mode(&self) -> DevTimeMode;
    pub fn skip_by(&self, ms: u64) -> u64 /* skipped */; pub fn skip_to(&self, target_ms: u64) -> u64 /* 0 if target <= now */;
    pub async fn skip_guarded<F: FnOnce(u64 /*now*/) -> Option<u64 /*target*/>>(&self, check: F) -> u64; // write-locks the barrier, re-checks, skips
    pub fn barrier(&self) -> Arc<tokio::sync::RwLock<()>>;
}
impl ostro_partition::clock::Clock for DevClock { fn now_ms(&self) -> u64 { self.base.now_ms() + self.offset_ms() } fn subscribe_changes(&self) -> Option<watch::Receiver<u64>> { Some(self.tx.subscribe()) } fn skip_barrier(&self) -> Option<Arc<tokio::sync::RwLock<()>>> { Some(self.barrier()) } }
impl ostro_engine::env::Clock for DevClock { fn now_unix_nanos(&self) -> u64 { ostro_partition::clock::Clock::now_ms(self) * 1_000_000 + sub-ms from base? (use ms * 1e6; document) } }
```
