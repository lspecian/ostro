Contract to create EXACTLY (04-16 implements it):
```rust
pub struct Job {
    pub partition: PartitionId,
    pub run: RunId,
    pub token: AttemptToken,
    pub record: RunRecord,                                    // read right after the grant
    pub handle: PartitionHandle<Rf1Log>,
    pub inputs: tokio::sync::mpsc::UnboundedReceiver<u64>,    // seqs of inputs journaled while leased (InputAppended)
    pub revoked: tokio::sync::watch::Receiver<bool>,          // true = lease lost (expiry / shutdown): stop now
    pub lease: LeaseRenewer,                                  // renew() -> bool (false once revoked)
    pub failures: u32,                                        // consecutive AttemptFailed outcomes of this run (in memory)
}
pub enum ExecOutcome {
    Released,                 // terminal, suspended or requeued by the executor's own command: drop lease
    Reactivate,               // run again now with the SAME token (hot-wait elapsed, append conflict, budget pressure)
    Retry { after_ms: u64 },  // attempt failed retryably: grant a NEW attempt after the delay
    Fenced,                   // a newer attempt owns the run
}
pub trait Executor: Send + Sync + 'static {
    fn execute(&self, job: Job) -> std::pin::Pin<Box<dyn std::future::Future<Output = ExecOutcome> + Send + 'static>>;
}
pub struct DispatcherConfig { pub executor: Arc<dyn Executor>, pub lease_ttl: Duration, pub slots: Arc<tokio::sync::Semaphore>, pub stats: Option<Arc<DispatcherStats>> }
pub struct DispatcherStats { pub dispatched: AtomicU64, pub rescans: AtomicU64, pub lease_expired: AtomicU64, pub reactivated: AtomicU64 }
// LeaderConfig gains: pub dispatcher: Option<DispatcherConfig>  (manual Debug)
```
