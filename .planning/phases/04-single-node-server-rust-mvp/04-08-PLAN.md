```rust
// crates/ostro-api/src/domain.rs
#[async_trait::async_trait]
pub trait ControlPlane: Send + Sync + 'static {
    async fn upload_module(&self, req: UploadModule) -> Result<UploadModuleResult, ApiError>;
    async fn get_module(&self, sha256: String) -> Result<ModuleInfo, ApiError>;
    async fn list_workflows(&self) -> Result<Vec<WorkflowInfo>, ApiError>;
    async fn activate_workflow(&self, req: ActivateWorkflow) -> Result<WorkflowInfo, ApiError>;
    async fn start_run(&self, req: StartRunRequest) -> Result<StartRunResult, ApiError>;
    async fn get_run(&self, run_id: String) -> Result<RunInfo, ApiError>;
    async fn run_history(&self, req: HistoryRequest) -> Result<HistoryPage, ApiError>;
    async fn list_runs(&self, req: ListRunsRequest) -> Result<RunPage, ApiError>;
    /// Meta has a leader and every partition slot is ready.
    fn ready(&self) -> bool;
}
pub struct UploadModule { pub sha256: String /*lowercase hex*/, pub wasm: bytes::Bytes, pub language: Option<String> }
pub struct UploadModuleResult { pub module: ModuleInfo, pub created: bool }
pub struct WorkflowRef { pub name: String, pub version: u32 }
pub struct ModuleInfo { pub sha256: String, pub size_bytes: u64, pub language: Option<String>, pub workflows: Vec<WorkflowRef>, pub uploaded_at_ms: u64 }
pub struct WorkflowVersionInfo { pub version: u32, pub module_sha256: String, pub default_queue: Option<String>, pub run_timeout_ms: Option<u64> }
pub struct WorkflowInfo { pub name: String, pub active_version: Option<u32>, pub versions: Vec<WorkflowVersionInfo>, pub record_version: u64 }
pub struct ActivateWorkflow { pub name: String, pub version: u32, pub expected_version: Option<u64> }
pub enum ConflictPolicy { ReturnExisting /*default*/, Reject }
pub struct StartRunRequest { pub workflow: String, pub input_json: Vec<u8> /*validated JSON text*/, pub version: Option<u32>, pub queue: Option<String>, pub delay_ms: Option<u64>, pub priority: Option<i32>, pub tags: Vec<String>, pub traceparent: Option<String>, pub tenant: Option<String>, pub idempotency_key: Option<String>, pub conflict_policy: ConflictPolicy, pub concurrency_key: Option<String> }
pub struct StartRunResult { pub run_id: String, pub created: bool }
pub enum RunStatus { Scheduled, Ready, Running, Completed, Failed, Cancelled }
pub struct RunInfo { pub run_id: String, pub workflow: String, pub version: u32, pub module_sha256: String, pub queue: String, pub status: RunStatus, pub attempt: Option<u32>, pub tags: Vec<String>, pub tenant: Option<String>, pub traceparent: Option<String>, pub priority: i32, pub concurrency_key: Option<String>, pub created_at_ms: u64, pub completed_at_ms: Option<u64>, pub published_status_json: Option<Vec<u8>>, pub result_json: Option<Vec<u8>>, pub failure: Option<String> }
pub struct HistoryRequest { pub run_id: String, pub from_seq: u64, pub limit: u32 }
pub struct HistoryEvent { pub seq: u64, pub kind: String, pub ts_ms: u64, pub summary: String, pub details_json: Vec<u8> }
pub struct HistoryPage { pub events: Vec<HistoryEvent>, pub next_from_seq: Option<u64> }
pub struct ListRunsRequest { pub workflow: Option<String>, pub status: Option<RunStatus>, pub queue: Option<String>, pub tag: Option<String>, pub created_after_ms: Option<u64>, pub created_before_ms: Option<u64>, pub limit: u32, pub page_token: Option<String> }
pub struct RunSummary { pub run_id: String, pub workflow: String, pub version: u32, pub status: RunStatus, pub queue: String, pub created_at_ms: u64, pub tags: Vec<String> }
pub struct RunPage { pub runs: Vec<RunSummary>, pub next_page_token: Option<String> }
// crates/ostro-api/src/error.rs
pub enum ErrorCode { InvalidArgument, NotFound, Conflict, PreconditionFailed, PayloadTooLarge, Unavailable, Internal }
pub struct ApiError { pub code: ErrorCode, pub message: String, pub details: Option<serde_json::Value> }
impl ApiError { pub fn http_status(&self) -> http::StatusCode; pub fn grpc_code(&self) -> tonic::Code; }
// lib.rs
pub struct ApiConfig { pub max_module_bytes: usize, pub max_request_bytes: usize }
pub fn router(cp: Arc<dyn ControlPlane>, cfg: ApiConfig) -> axum::Router;   // REST + /openapi.json + health + UI (gRPC merged by 04-11)
pub fn openapi() -> utoipa::openapi::OpenApi;                                // used by `ostro openapi`
```
